rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // جدول المنافسة: أي لاعب داخل بجوجل يقرا، وكل لاعب يكتب صفّه بس
    match /players/{uid} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == uid;
    }
    // سوق اللاعيبة: الكل يقرا، صاحب العرض بس ينشره، والقبول بيكون باسم اللي قبله
    match /market/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.owner == request.auth.uid;
      allow update: if request.auth != null && (resource.data.owner == request.auth.uid || resource.data.taker == request.auth.uid || (resource.data.status == 'open' && request.resource.data.taker == request.auth.uid && (!('to' in resource.data) || resource.data.to == request.auth.uid)));
      allow delete: if request.auth != null && resource.data.owner == request.auth.uid;
    }
    // أكواد الحسابات البنكية: أي لاعب يدوّر على كود، وصاحب الحساب بس يسجّله
    match /accounts/{code} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null && request.resource.data.owner == request.auth.uid && (resource == null || resource.data.owner == request.auth.uid);
      allow delete: if request.auth != null && resource.data.owner == request.auth.uid;
    }
    // التحويلات بين اللاعيبة: الراسل والمستلم بس يشوفوها
    match /transfers/{id} {
      allow read: if request.auth != null && (resource.data.to == request.auth.uid || resource.data.from == request.auth.uid);
      allow create: if request.auth != null && request.resource.data.from == request.auth.uid && request.resource.data.usd > 0 && request.resource.data.status == 'sent';
      allow update: if request.auth != null && resource.data.to == request.auth.uid && request.resource.data.usd == resource.data.usd && request.resource.data.to == resource.data.to;
    }
    // الشات العام
    match /chat/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid && request.resource.data.text is string && request.resource.data.text.size() <= 300;
      allow delete: if request.auth != null && resource.data.uid == request.auth.uid;
    }
    // الصداقة: الطرفين بس
    match /freq/{id} {
      allow read, update, delete: if request.auth != null && request.auth.uid in resource.data.users;
      allow create: if request.auth != null && request.resource.data.from == request.auth.uid && request.auth.uid in request.resource.data.users && request.resource.data.users.size() == 2;
    }
    // حفظ اللعبة: كل لاعب يقرا ويكتب حفظه بس
    match /data/users/{uid}/{rest=**} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
  }
}
